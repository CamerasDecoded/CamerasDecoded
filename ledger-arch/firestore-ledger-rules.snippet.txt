// Design notes:
// - Deterministic instructorLinks IDs ({iid}_{sid}) let rules do a single
//   get() — rules cannot query, ever.
// - Instructor reads are FAIL-CLOSED: the query MUST filter tier == 'teaching'.
//   Without it, resource.data.tier == 'teaching' fails on platform docs and
//   the entire query is denied. This is correct behavior, not a bug.
// - Temporal bounds (§2.5a): instructors see only events within their
//   enrollment window (enrolledAt .. accessExpiresAt). Enforced per-document.
// - Admin bypass uses custom claims (admin: true). Migrate from users.role
//   via the one-time backfill before enabling these rules.
// - Service account (Cloud Functions) bypasses rules for rollups, skillProfile,
//   homework verification, and backfills.

function cdIsSignedIn() { return request.auth != null; }
function cdIsAdmin() {
  return cdIsSignedIn() && request.auth.token.admin == true;
}
function cdIsInstructorRole() {
  return cdIsSignedIn() && request.auth.token.instructor == true;
}
function cdLinkId(iid, sid) { return iid + '_' + sid; }
function cdActiveLink(iid, sid) {
  let link = get(/databases/$(database)/documents/instructorLinks/$(cdLinkId(iid, sid)));
  return link != null
      && link.data.status == 'active'
      && link.data.instructorUid == iid
      && link.data.studentUid == sid;
}
// Temporal bound check: event ts must fall inside the instructor's window.
function cdInWindow(iid, sid, eventTs) {
  let link = get(/databases/$(database)/documents/instructorLinks/$(cdLinkId(iid, sid)));
  return link != null
      && link.data.status == 'active'
      && eventTs >= link.data.enrolledAt
      && (link.data.accessExpiresAt == null || eventTs <= link.data.accessExpiresAt);
}
function cdCanSeeTeaching(uid) {
  return cdIsSignedIn() && (
    request.auth.uid == uid || cdIsAdmin() ||
    (cdIsInstructorRole() && cdActiveLink(request.auth.uid, uid))
  );
}

// ---- THE LEDGER: users/{uid}/ledger/{eventId} ----
match /users/{uid}/ledger/{eventId} {
  // Read: owner, admin, or active instructor within their temporal window.
  // Instructors ONLY see teaching tier (fail-closed on the tier filter).
  allow read: if cdIsSignedIn() && (
    request.auth.uid == uid || cdIsAdmin() ||
    (cdIsInstructorRole()
      && cdInWindow(request.auth.uid, uid, resource.data.ts)
      && resource.data.tier == 'teaching')
  );
  // Create-only, schema-validated. No client updates or deletes.
  allow create: if cdIsSignedIn() && request.auth.uid == uid
    && request.resource.data.uid == uid
    && request.resource.data.tier in ['teaching', 'platform']
    && request.resource.data.type in [
         // Teaching tier (spec Appendix B)
         'fm.lesson.completed', 'fm.quiz.attempted', 'fm.chapter.completed',
         'fm.exam.attempted', 'arcade.session.scored', 'arcade.shot.graded',
         'arena.entry.submitted', 'darkroom.lesson.completed',
         'darkroom.drill.completed', 'badge.earned', 'streak.milestone',
         'homework.assigned', 'homework.completed', 'user.enrolled',
         // Platform tier (spec Appendix B)
         'fm.lesson.started', 'arcade.session.started', 'user.signup',
         'tier.changed', 'notification.interaction', 'ai.workbench.used',
         'referral.completed', 'arena.round.swept', 'device.registered',
         'system.xp.adjusted', 'arena.vote.cast']
    && request.resource.data.ts is timestamp
    && request.resource.data.clientTs is number
    && request.resource.data.day is string
    && request.resource.data.xp is number
    && request.resource.data.xp >= -1000 && request.resource.data.xp <= 500
    && request.resource.data.detail is map
    && request.resource.data.schemaV == 1;
  allow update, delete: if cdIsAdmin();
}

// ---- ROLLUPS & SKILL PROFILE (server-written only) ----
match /users/{uid}/ledgerDaily/{day} {
  allow read: if cdCanSeeTeaching(uid);
  allow write: if false; // Cloud Function (service account) only
}
match /users/{uid}/skillProfile {
  allow read: if cdCanSeeTeaching(uid);
  allow write: if false; // Cloud Function (service account) only
}

// ---- INSTRUCTOR LINKS (invite-link model, spec §3.2) ----
match /instructorLinks/{lid} {
  // Both parties + admin can read (student transparency requirement).
  allow read: if cdIsSignedIn() && (
    request.auth.uid == resource.data.instructorUid ||
    request.auth.uid == resource.data.studentUid || cdIsAdmin());
  // Created at signup via invite link: the signup flow (server or client
  // with the invite code) creates the doc naming the link's instructor.
  // Doc ID must be the deterministic composite.
  allow create: if cdIsSignedIn()
    && lid == cdLinkId(request.resource.data.instructorUid,
                       request.resource.data.studentUid)
    && request.resource.data.status == 'active'
    && request.resource.data.enrolledAt is timestamp;
  // Either party revokes; admin manages. Revocation sets endedAt +
  // accessExpiresAt (endedAt + 24h) — enforced by the writer, read-checked above.
  allow update: if cdIsSignedIn() && (
    (request.auth.uid == resource.data.studentUid &&
     request.resource.data.status == 'revoked') ||
    (request.auth.uid == resource.data.instructorUid &&
     request.resource.data.status == 'revoked') || cdIsAdmin());
  allow delete: if cdIsAdmin();
}

// ---- HOMEWORK ----
match /users/{uid}/homework/{hwId} {
  allow read: if cdCanSeeTeaching(uid);
  allow create: if cdIsInstructorRole() && cdActiveLink(request.auth.uid, uid)
    && request.resource.data.assignedBy == request.auth.uid;
  allow update: if false; // verifier function only
  allow delete: if cdIsAdmin();
}

// ---- ADMIN AUDIT (append-only) ----
match /adminAudit/{auditId} {
  allow read: if cdIsAdmin();
  allow create: if cdIsAdmin() && request.resource.data.adminUid == request.auth.uid
    && request.resource.data.ts is timestamp;
  allow update, delete: if false;
}
